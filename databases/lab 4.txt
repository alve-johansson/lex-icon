SELECT c.first_name, c.last_name, o.status
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id;

SELECT c.first_name, c.last_name, o.status
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
WHERE c.first_name = 'Erik'

SELECT c.first_name, c.last_name, o.status
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
WHERE c.city = 'Göteborg';

SELECT o.order_id, p.name, p.category
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id;

SELECT o.order_id, p.name
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
WHERE p.category = 'Shoes';

SELECT o.order_id, p.name, p.price, oi.quantity, (p.price * oi.quantity) AS line_total
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
WHERE o.order_id = 10;

SELECT o.order_id, c.first_name, p.name, p.price, oi.quantity, o.order_date
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
JOIN customers c ON o.customer_id = c.customer_id
WHERE p.name = 'Hoodie Black';

SELECT o.order_id, c.first_name, c.last_name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.customer_id;

SELECT p.name
FROM products p
LEFT JOIN order_items oi ON p.product_id = oi.product_id
WHERE oi.product_id IS NULL;

SELECT o.order_id, c.first_name, p.name, oi.quantity
FROM customers c
JOIN orders o ON o.customer_id = c.customer_id
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
WHERE c.city = 'Uppsala';
