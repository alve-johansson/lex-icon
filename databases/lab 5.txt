SELECT COUNT (*)
FROM orders;

SELECT *
FROM products
WHERE (category = 'Clothing' OR category = 'Accessories')
AND price BETWEEN 150 AND 500
ORDER BY price DESC;

SELECT *
FROM orders
WHERE order_date LIKE '2026-02%' AND status NOT LIKE 'cancelled';

SELECT o.order_id, p.name, oi.quantity, (p.price * oi.quantity) AS line_total
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
JOIN products p ON oi.product_id = p.product_id
WHERE line_total > 500
ORDER BY line_total DESC;

SELECT DISTINCT c.first_name, c.last_name
FROM customers c
JOIN orders o ON o.customer_id = c.customer_id
WHERE city = 'Uppsala' OR city = 'Stockholm';
